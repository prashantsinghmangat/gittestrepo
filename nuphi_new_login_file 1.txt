import { Component, OnInit } from '@angular/core';
import { ActivatedRoute, Router } from '@angular/router';
import { GlobalProfileService } from 'src/app/global-profile.service';
import { AuthService, User, OpenService, LoginRequest } from 'src/swagger';
import { CommonSupportService } from 'src/app/common-support.service';
declare var swal: any;

@Component({
  selector: 'app-login',
  templateUrl: './login.component.html',
  styleUrls: ['./login.component.css']
})
export class LoginComponent implements OnInit {

  public emailId: string;
  public password: string;
  public error: string;
  captchSiteKey = '6LdgoIUqAAAAANPSQNdncrDo9dmX5tziDfWGy04N';
  public loginDetails: LoginRequest = {}
  token: string;

  constructor(
    private activatedRoute: ActivatedRoute,
    public profileSync: GlobalProfileService,
    private authApi: AuthService,
    public router: Router,
    public openApi: OpenService,
    private commonService: CommonSupportService

  ) { }

  ngOnInit() {
    this.activatedRoute.params.subscribe(params => {
      if (params && params.token) {
        this.verifyUser(params.token);
      }
    });
  }

  public Login() {

    let redirectTo;

    if (this.activatedRoute.snapshot.queryParams.hasOwnProperty('redirectTo')) {
      redirectTo = this.activatedRoute.snapshot.queryParams.redirectTo;
    } else {
      redirectTo = null;
    }

    if (this.loginDetails) {
      this.error = null;
      const that = this;
      swal({
        title: 'Logging you in',
        html: '<i class="fa fa-circle-check fa-spin"></i>',
        type: 'info',
        showConfirmButton: false,
        showCancelButton: false,
      });

      this.profileSync.updateUserProfile(null);
      this.profileSync.updateLoggedInStatus(false);
      grecaptcha.ready(() => {
        grecaptcha.execute(this.captchSiteKey, { action: 'login' }).then((token: string) => {
          this.token = token;
          return new Promise((resolve, reject) => {
            console.log(this.token);
            //debugger
            that.openApi.login(this.loginDetails, this.token).subscribe(user => {
              // remove this code while go live on UAT, QA and PROD
              localStorage.setItem("USERINFO", JSON.stringify(user['result']));
              resolve(user);
            }, err => {
              console.log(err);
              reject(err);
            });
          }).then((user: User) => {
            user = user['result'];
            this.profileSync.setToken(user['access_token']);
            swal.close();
            const that = this;
            that.openApi.credlixProfile().subscribe(user => {
              if (user && user['result'] != null) {
                this.profileSync.updateUserProfile(user['result']);
                user = user['result'];
                // localStorage.setItem("ADMININFO", JSON.stringify(user));
                let name = user.loggedInUserRole.role.name;
                let roleType = user.loggedInUserRole.roleType
                if ((name == User.RoleEnum.COMPANY_ADMIN.toString() || name == User.RoleEnum.MAKER_ADMIN.toString() || name == User.RoleEnum.CHECKER_ADMIN.toString() || name == User.RoleEnum.VIEW_ADMIN.toString() || name == User.RoleEnum.SALESVIEW.toString()) && roleType == "EXIM") {
                  that.router.navigate(['/admin']);
                } else if (name == User.RoleEnum.VIEW_ADMIN.toString() && roleType == 'EXIM') {
                  that.router.navigate(['/view-admin']);
                } else if ((name == User.RoleEnum.COMPANY_ADMIN.toString() || name == User.RoleEnum.MAKER_ADMIN.toString()) && roleType == "BORROWER") {
                  that.router.navigate(['/exporter']);
                } else if (name == User.RoleEnum.FRESH_SUPPLIER.toString() && roleType == "CREDLIX") {
                  let id = user.userOnboardId;
                  that.router.navigate(['/supplier-new-dashboard', id]);//change for supplier on board
                }
                else {
                  that.router.navigate(['/login']);
                }
              } else {
                console.log('Failed to get user profile');
              }
            });

          }).catch(err => {
            swal.close();
            this.error = err.error ? err.error.message : err;
          });
        });
      });
    }
  }
  verifyUser(token) {
    this.openApi.verifyUser(token).subscribe(ok => {
      if (ok) {
        this.commonService.showSuccessSwal('Verified', ok.message);
      }
    }, err => {
      this.commonService.showErrorSwal(err);
      console.log(err);
    });
  }
}

  public login(
    LoginRequest: LoginRequest,
    captchaToken: any,
    observe: any = "body",
    reportProgress: boolean = false
  ): Observable<any> {
    if (LoginRequest === null || LoginRequest === undefined) {
      throw new Error(
        "Required parameter LoginRequest was null or undefined when calling login."
      );
    }

    let headers = this.defaultHeaders;
    // to determine the Accept header
    let httpHeaderAccepts: string[] = ["application/json"];
    const httpHeaderAcceptSelected: string | undefined =
      this.configuration.selectHeaderAccept(httpHeaderAccepts);
    if (httpHeaderAcceptSelected != undefined) {
      headers = headers.set("Accept", httpHeaderAcceptSelected);
    }

    // to determine the Content-Type header
    const consumes: string[] = ["application/json"];
    const httpContentTypeSelected: string | undefined =
      this.configuration.selectHeaderContentType(consumes);
    if (httpContentTypeSelected != undefined) {
      headers = headers.set("Content-Type", httpContentTypeSelected);
    }
    if(captchaToken){
      headers = headers.set("g-recaptcha-response", captchaToken.toString());
    }
    console.log(captchaToken);
    console.log(headers);
    return this.httpClient.post<User>(
      this.env + `/users/signin`,
      LoginRequest,
      {
        withCredentials: this.configuration.withCredentials,
        headers: headers,
        observe: observe,
        reportProgress: reportProgress,
      }
    );
  }
